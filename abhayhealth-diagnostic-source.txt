package com.example.abhayhealth.data.repository

import android.content.Context
import android.app.Activity
import com.example.abhayhealth.core.security.SecurityAudit
import com.example.abhayhealth.domain.model.AuthState
import com.example.abhayhealth.domain.model.AuthUser
import com.example.abhayhealth.domain.repository.AuthRepository
import com.google.firebase.FirebaseApp
import com.google.firebase.FirebaseException
import com.google.firebase.FirebaseTooManyRequestsException
import com.google.firebase.auth.FirebaseAuth
import com.google.firebase.auth.FirebaseAuthInvalidCredentialsException
import com.google.firebase.auth.FirebaseAuthRecentLoginRequiredException
import com.google.firebase.auth.PhoneAuthCredential
import com.google.firebase.auth.PhoneAuthOptions
import com.google.firebase.auth.PhoneAuthProvider
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.concurrent.TimeUnit

/**
 * Firebase implementation of AuthRepository for Phone OTP Authentication.
 */
class FirebaseAuthRepository(
    private val appContext: Context
) : AuthRepository {

    private val _authState = MutableStateFlow<AuthState>(AuthState.Initial)
    override val authState: StateFlow<AuthState> = _authState.asStateFlow()

    private var firebaseAuth: FirebaseAuth? = null

    init {
        try {
            val app = FirebaseApp.getApps(appContext)
                .firstOrNull { it.name == FirebaseApp.DEFAULT_APP_NAME }
                ?: FirebaseApp.initializeApp(appContext)
                ?: throw IllegalStateException(
                    "Default FirebaseApp is missing. Check google-services.json and generated Firebase resources."
                )

            firebaseAuth = FirebaseAuth.getInstance(app)

            firebaseAuth?.addAuthStateListener { firebase ->
                val user = firebase.currentUser

                if (user != null) {
                    _authState.value = AuthState.Authenticated(
                        AuthUser(
                            uid = user.uid,
                            phoneNumber = user.phoneNumber
                        )
                    )
                } else if (
                    _authState.value !is AuthState.CodeSent &&
                    _authState.value !is AuthState.Loading
                ) {
                    _authState.value = AuthState.Unauthenticated
                }
            }
        } catch (e: Exception) {
            SecurityAudit.logError(
                "FirebaseAuthRepository",
                "Firebase initialization exception",
                e
            )

            _authState.value = AuthState.Error(
                message = "Firebase initialization failed: ${
                    e.localizedMessage ?: "Unknown initialization error"
                }",
                isConfigurationError = true
            )
        }
    }

    override val currentUser: AuthUser?
        get() = firebaseAuth?.currentUser?.let { user ->
            AuthUser(uid = user.uid, phoneNumber = user.phoneNumber)
        }

    override val isAuthenticated: Boolean
        get() = firebaseAuth?.currentUser != null

    override fun sendOtp(phoneNumber: String, activity: Activity) {
        val auth = firebaseAuth
        if (auth == null) {
            _authState.value = AuthState.Error(
                message = "Firebase Authentication is not configured. google-services.json required.",
                isConfigurationError = true
            )
            return
        }

        _authState.value = AuthState.Loading

        val callbacks = object : PhoneAuthProvider.OnVerificationStateChangedCallbacks() {
            override fun onVerificationCompleted(credential: PhoneAuthCredential) {
                SecurityAudit.logEvent("Phone verification automatically completed by system")
                signInWithCredential(credential)
            }

            override fun onVerificationFailed(e: FirebaseException) {
                SecurityAudit.logError("FirebaseAuthRepository", "Phone verification failed", e)
                val errorMessage = when (e) {
                    is FirebaseTooManyRequestsException -> "Quota exceeded or too many requests. Please try again later."
                    is FirebaseAuthInvalidCredentialsException -> "Invalid phone number format provided."
                    else -> e.localizedMessage ?: "Verification failed. Please check your network and try again."
                }
                _authState.value = AuthState.Error(errorMessage)
            }

            override fun onCodeSent(verificationId: String, token: PhoneAuthProvider.ForceResendingToken) {
                SecurityAudit.logEvent("OTP code sent successfully")
                _authState.value = AuthState.CodeSent(
                    verificationId = verificationId,
                    phoneNumber = phoneNumber,
                    resendToken = token
                )
            }
        }

        val options = PhoneAuthOptions.newBuilder(auth)
            .setPhoneNumber(phoneNumber)
            .setTimeout(60L, TimeUnit.SECONDS)
            .setActivity(activity)
            .setCallbacks(callbacks)
            .build()

        PhoneAuthProvider.verifyPhoneNumber(options)
    }

    override fun verifyOtp(verificationId: String, code: String) {
        if (code.isBlank() || code.length < 6) {
            _authState.value = AuthState.Error("Please enter the complete 6-digit OTP code.")
            return
        }

        _authState.value = AuthState.Loading
        try {
            val credential = PhoneAuthProvider.getCredential(verificationId, code)
            signInWithCredential(credential)
        } catch (e: Exception) {
            SecurityAudit.logError("FirebaseAuthRepository", "Invalid credential error", e)
            _authState.value = AuthState.Error("Invalid code entered. Please verify and try again.")
        }
    }

    override fun resendOtp(phoneNumber: String, activity: Activity, resendToken: Any?) {
        val auth = firebaseAuth
        if (auth == null) {
            _authState.value = AuthState.Error("Firebase Auth is not configured.", isConfigurationError = true)
            return
        }

        _authState.value = AuthState.Loading

        val callbacks = object : PhoneAuthProvider.OnVerificationStateChangedCallbacks() {
            override fun onVerificationCompleted(credential: PhoneAuthCredential) {
                signInWithCredential(credential)
            }

            override fun onVerificationFailed(e: FirebaseException) {
                SecurityAudit.logError("FirebaseAuthRepository", "Resend OTP failed", e)
                _authState.value = AuthState.Error(e.localizedMessage ?: "Failed to resend OTP.")
            }

            override fun onCodeSent(verificationId: String, token: PhoneAuthProvider.ForceResendingToken) {
                SecurityAudit.logEvent("OTP code resent successfully")
                _authState.value = AuthState.CodeSent(
                    verificationId = verificationId,
                    phoneNumber = phoneNumber,
                    resendToken = token
                )
            }
        }

        val builder = PhoneAuthOptions.newBuilder(auth)
            .setPhoneNumber(phoneNumber)
            .setTimeout(60L, TimeUnit.SECONDS)
            .setActivity(activity)
            .setCallbacks(callbacks)

        if (resendToken is PhoneAuthProvider.ForceResendingToken) {
            builder.setForceResendingToken(resendToken)
        }

        PhoneAuthProvider.verifyPhoneNumber(builder.build())
    }

    private fun signInWithCredential(credential: PhoneAuthCredential) {
        val auth = firebaseAuth ?: return
        auth.signInWithCredential(credential)
            .addOnCompleteListener { task ->
                if (task.isSuccessful) {
                    val user = task.result?.user
                    if (user != null) {
                        _authState.value = AuthState.Authenticated(
                            AuthUser(uid = user.uid, phoneNumber = user.phoneNumber)
                        )
                    } else {
                        _authState.value = AuthState.Error("Sign-in succeeded but user object was null.")
                    }
                } else {
                    val exception = task.exception
                    SecurityAudit.logError("FirebaseAuthRepository", "Sign-in with credential failed", exception)
                    val errorMsg = when (exception) {
                        is FirebaseAuthInvalidCredentialsException -> "Incorrect or expired OTP entered."
                        else -> exception?.localizedMessage ?: "Authentication failed. Please try again."
                    }
                    _authState.value = AuthState.Error(errorMsg)
                }
            }
    }

    override fun signOut() {
        try {
            firebaseAuth?.signOut()
            _authState.value = AuthState.Unauthenticated
        } catch (e: Exception) {
            SecurityAudit.logError("FirebaseAuthRepository", "Sign-out error", e)
            _authState.value = AuthState.Unauthenticated
        }
    }

    override fun deleteAccount(onComplete: (Boolean, String?) -> Unit) {
        val user = firebaseAuth?.currentUser
        if (user != null) {
            user.delete().addOnCompleteListener { task ->
                if (task.isSuccessful) {
                    _authState.value = AuthState.Unauthenticated
                    onComplete(true, null)
                } else {
                    val exception = task.exception
                    val errorMsg = if (exception is FirebaseAuthRecentLoginRequiredException) {
                        "Security requirement: Recent login required before deleting account. Please sign out and sign in again."
                    } else {
                        exception?.localizedMessage ?: "Failed to delete account."
                    }
                    onComplete(false, errorMsg)
                }
            }
        } else {
            _authState.value = AuthState.Unauthenticated
            onComplete(true, null)
        }
    }
}

package com.example.abhayhealth.data.repository

import android.content.Context
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey
import com.example.abhayhealth.core.storage.datastore.dataStore
import com.example.abhayhealth.domain.model.PatientProfile
import com.example.abhayhealth.domain.repository.PatientProfileRepository
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.flow.map

class PatientProfileRepositoryImpl(
    private val context: Context
) : PatientProfileRepository {

    private val activeProfileFlow = MutableStateFlow<PatientProfile?>(null)

    private fun getProfileKey(accountUid: String) = stringPreferencesKey("profile_data_$accountUid")

    override fun getProfile(accountUid: String): Flow<PatientProfile?> {
        val key = getProfileKey(accountUid)
        return context.dataStore.data.map { prefs ->
            val serialized = prefs[key]
            if (serialized != null && serialized.isNotBlank()) {
                deserializeProfile(serialized, accountUid)
            } else {
                null
            }
        }
    }

    override suspend fun saveProfile(profile: PatientProfile) {
        val key = getProfileKey(profile.accountUid)
        val serialized = serializeProfile(profile)
        context.dataStore.edit { prefs ->
            prefs[key] = serialized
        }
        activeProfileFlow.value = profile
    }

    override suspend fun deleteProfile(accountUid: String) {
        if (accountUid.isBlank()) return
        val key = getProfileKey(accountUid)
        context.dataStore.edit { prefs ->
            prefs.remove(key)
        }
        if (activeProfileFlow.value?.accountUid == accountUid) {
            activeProfileFlow.value = null
        }
    }

    override suspend fun hasProfile(accountUid: String): Boolean {
        if (accountUid.isBlank()) return false
        val key = getProfileKey(accountUid)
        val prefs = context.dataStore.data.first()
        val serialized = prefs[key]
        return serialized != null && serialized.isNotBlank()
    }

    override suspend fun clearCurrentProfileMemory() {
        activeProfileFlow.value = null
    }

    private fun serializeProfile(profile: PatientProfile): String {
        return listOf(
            profile.fullName,
            profile.dateOfBirth ?: "",
            profile.gender ?: "",
            profile.city ?: "",
            profile.state ?: "",
            profile.preferredLanguage,
            profile.photoUri ?: "",
            profile.bloodGroup ?: "",
            profile.allergies ?: "",
            profile.healthConditions ?: "",
            profile.emergencyContactName ?: "",
            profile.emergencyContactPhone ?: "",
            profile.additionalNotes ?: "",
            profile.updatedAtTimestamp.toString()
        ).joinToString("|||")
    }

    private fun deserializeProfile(serialized: String, accountUid: String): PatientProfile {
        val parts = serialized.split("|||")
        return PatientProfile(
            accountUid = accountUid,
            fullName = parts.getOrNull(0) ?: "",
            dateOfBirth = parts.getOrNull(1).takeIf { !it.isNull_or_blank() },
            gender = parts.getOrNull(2).takeIf { !it.isNull_or_blank() },
            city = parts.getOrNull(3).takeIf { !it.isNull_or_blank() },
            state = parts.getOrNull(4).takeIf { !it.isNull_or_blank() },
            preferredLanguage = parts.getOrNull(5).takeIf { !it.isNull_or_blank() } ?: "English",
            photoUri = parts.getOrNull(6).takeIf { !it.isNull_or_blank() },
            bloodGroup = parts.getOrNull(7).takeIf { !it.isNull_or_blank() },
            allergies = parts.getOrNull(8).takeIf { !it.isNull_or_blank() },
            healthConditions = parts.getOrNull(9).takeIf { !it.isNull_or_blank() },
            emergencyContactName = parts.getOrNull(10).takeIf { !it.isNull_or_blank() },
            emergencyContactPhone = parts.getOrNull(11).takeIf { !it.isNull_or_blank() },
            additionalNotes = parts.getOrNull(12).takeIf { !it.isNull_or_blank() },
            updatedAtTimestamp = parts.getOrNull(13)?.toLongOrNull() ?: System.currentTimeMillis()
        )
    }
}

private fun String?.isNull_or_blank(): Boolean = this == null || this.isBlank()

/**

 */
@Dao
interface MedicalRecordDao {

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    fun insertRecord(record: MedicalRecord): Long

    @Query("SELECT * FROM medical_records WHERE accountUid = :accountUid ORDER BY recordDate DESC")
    fun getRecordsByAccount(accountUid: String): Flow<List<MedicalRecord>>

    @Query("SELECT * FROM medical_records WHERE accountUid = :accountUid AND (title LIKE '%' || :query || '%' OR category LIKE '%' || :query || '%') ORDER BY recordDate DESC")
    fun searchRecords(accountUid: String, query: String): Flow<List<MedicalRecord>>

    @Query("SELECT * FROM medical_records WHERE id = :id AND accountUid = :accountUid")
    fun getRecordById(id: Long, accountUid: String): MedicalRecord?

    @Delete
    fun deleteRecord(record: MedicalRecord): Int

    @Query("DELETE FROM medical_records WHERE id = :id AND accountUid = :accountUid")
    fun deleteRecordById(id: Long, accountUid: String): Int
}

package com.example.abhayhealth.feature.records

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import androidx.room.migration.Migration
import androidx.sqlite.db.SupportSQLiteDatabase

/**
 * Room Database for local medical record metadata and extracted information.
 */
@Database(
    entities = [MedicalRecord::class, ExtractedMedicalInfo::class],
    version = 2,
    exportSchema = false
)
abstract class MedicalRecordDatabase : RoomDatabase() {

    abstract fun medicalRecordDao(): MedicalRecordDao
    abstract fun extractedInfoDao(): ExtractedInfoDao

    companion object {
        const val DATABASE_NAME = "abhayhealth_medical_records.db"

        val MIGRATION_1_2 = object : Migration(1, 2) {
            override fun migrate(db: SupportSQLiteDatabase) {
                db.execSQL(
                    """
                    CREATE TABLE IF NOT EXISTS `extracted_medical_info` (
                        `recordId` INTEGER NOT NULL,
                        `accountUid` TEXT NOT NULL,
                        `rawText` TEXT NOT NULL,
                        `documentTitle` TEXT NOT NULL,
                        `reportDate` TEXT,
                        `structuredResultsJson` TEXT NOT NULL,
                        `reviewStatus` TEXT NOT NULL,
                        `processedPages` INTEGER NOT NULL,
                        `updatedAtTimestamp` INTEGER NOT NULL,
                        PRIMARY KEY(`recordId`),
                        FOREIGN KEY(`recordId`) REFERENCES `medical_records`(`id`) ON DELETE CASCADE
                    )
                    """.trimIndent()
                )
            }
        }

        @Volatile
        private var INSTANCE: MedicalRecordDatabase? = null

        fun getInstance(context: Context): MedicalRecordDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    MedicalRecordDatabase::class.java,
                    DATABASE_NAME
                )
                .addMigrations(MIGRATION_1_2)
                .build()
                INSTANCE = instance
                instance
            }
        }
    }
}

package com.example.abhayhealth.feature.records

import com.example.abhayhealth.core.common.DefaultDispatcherProvider
import com.example.abhayhealth.core.common.DispatcherProvider
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.withContext

/**
 * Repository for managing local medical record metadata and extracted information.
 */
class MedicalRecordRepository(
    private val medicalRecordDao: MedicalRecordDao,
    private val extractedInfoDao: ExtractedInfoDao,
    private val dispatcherProvider: DispatcherProvider = DefaultDispatcherProvider()
) {
    fun getRecords(accountUid: String): Flow<List<MedicalRecord>> {
        return medicalRecordDao.getRecordsByAccount(accountUid)
    }

    fun searchRecords(accountUid: String, query: String): Flow<List<MedicalRecord>> {
        return medicalRecordDao.searchRecords(accountUid, query)
    }

    suspend fun getRecordById(id: Long, accountUid: String): MedicalRecord? = withContext(dispatcherProvider.io) {
        medicalRecordDao.getRecordById(id, accountUid)
    }

    suspend fun insertRecord(record: MedicalRecord): Long = withContext(dispatcherProvider.io) {
        medicalRecordDao.insertRecord(record)
    }

    suspend fun deleteRecord(record: MedicalRecord) = withContext(dispatcherProvider.io) {
        medicalRecordDao.deleteRecord(record)
    }

    suspend fun deleteRecordById(id: Long, accountUid: String): Int = withContext(dispatcherProvider.io) {
        medicalRecordDao.deleteRecordById(id, accountUid)
    }

    // Extracted Information operations
    fun getExtractedInfo(recordId: Long, accountUid: String): Flow<ExtractedMedicalInfo?> {
        return extractedInfoDao.getExtractedInfo(recordId, accountUid)
    }

    suspend fun getExtractedInfoSync(recordId: Long, accountUid: String): ExtractedMedicalInfo? = withContext(dispatcherProvider.io) {
        extractedInfoDao.getExtractedInfoSync(recordId, accountUid)
    }

    suspend fun saveExtractedInfo(info: ExtractedMedicalInfo) = withContext(dispatcherProvider.io) {
        extractedInfoDao.insertOrUpdate(info)
    }

    suspend fun saveExtractedInfoWithTitleUpdate(
        info: ExtractedMedicalInfo,
        newTitle: String? = null
    ) = withContext(dispatcherProvider.io) {
        extractedInfoDao.insertOrUpdate(info)
        if (!newTitle.isNullOrBlank()) {
            val record = medicalRecordDao.getRecordById(info.recordId, info.accountUid)
            if (record != null && record.title != newTitle) {
                medicalRecordDao.insertRecord(record.copy(title = newTitle))
            }
        }
    }

    suspend fun deleteExtractedInfo(recordId: Long, accountUid: String) = withContext(dispatcherProvider.io) {
        extractedInfoDao.deleteExtractedInfo(recordId, accountUid)
    }
}

package com.example.abhayhealth.feature.records

import android.content.Context
import android.net.Uri
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.abhayhealth.core.common.DefaultDispatcherProvider
import com.example.abhayhealth.core.common.DispatcherProvider
import com.example.abhayhealth.domain.model.AuthState
import com.example.abhayhealth.domain.repository.AuthRepository
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class RecordsViewModel(
    private val medicalRecordRepository: MedicalRecordRepository? = null,
    private val authRepository: AuthRepository? = null,
    private val dispatcherProvider: DispatcherProvider = DefaultDispatcherProvider()
) : ViewModel() {

    private val _records = MutableStateFlow<List<MedicalRecord>>(emptyList())
    val records: StateFlow<List<MedicalRecord>> = _records.asStateFlow()

    private val _searchQuery = MutableStateFlow("")
    val searchQuery: StateFlow<String> = _searchQuery.asStateFlow()

    private val _isImporting = MutableStateFlow(false)
    val isImporting: StateFlow<Boolean> = _isImporting.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private val _infoMessage = MutableStateFlow<String?>(null)
    val infoMessage: StateFlow<String?> = _infoMessage.asStateFlow()

    private val _recordToDelete = MutableStateFlow<MedicalRecord?>(null)
    val recordToDelete: StateFlow<MedicalRecord?> = _recordToDelete.asStateFlow()

    private var observeJob: Job? = null

    init {
        observeAuthState()
    }

    private fun observeAuthState() {
        val authRepo = authRepository ?: return
        viewModelScope.launch(dispatcherProvider.main) {
            authRepo.authState.collectLatest { state ->
                when (state) {
                    is AuthState.Authenticated -> {
                        observeRecordsForUid(state.user.uid)
                    }
                    else -> {
                        clearAllState()
                    }
                }
            }
        }
    }

    fun observeRecords() {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty()) {
            clearAllState()
        } else {
            observeRecordsForUid(uid)
        }
    }

    private fun observeRecordsForUid(uid: String) {
        val repo = medicalRecordRepository ?: return
        observeJob?.cancel()
        observeJob = viewModelScope.launch(dispatcherProvider.main) {
            if (_searchQuery.value.isBlank()) {
                repo.getRecords(uid).collectLatest { list ->
                    _records.value = list
                }
            } else {
                repo.searchRecords(uid, _searchQuery.value).collectLatest { list ->
                    _records.value = list
                }
            }
        }
    }

    private fun clearAllState() {
        observeJob?.cancel()
        _records.value = emptyList()
        _searchQuery.value = ""
        _errorMessage.value = null
        _infoMessage.value = null
        _recordToDelete.value = null
    }

    fun onSearchQueryChanged(query: String) {
        _searchQuery.value = query
        val uid = authRepository?.currentUser?.uid ?: return
        observeRecordsForUid(uid)
    }

    fun clearMessages() {
        _errorMessage.value = null
        _infoMessage.value = null
    }

    fun importDocument(uri: Uri, context: Context) {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty()) {
            _errorMessage.value = "Please sign in to import medical documents."
            return
        }

        val repo = medicalRecordRepository
        if (repo == null) {
            _errorMessage.value = "Local database unavailable."
            return
        }

        viewModelScope.launch(dispatcherProvider.main) {
            _isImporting.value = true
            _errorMessage.value = null

            val result = withContext(Dispatchers.IO) {
                DocumentImportManager.importDocument(uri, context)
            }

            when (result) {
                is DocumentImportResult.Success -> {
                    val record = MedicalRecord(
                        accountUid = uid,
                        title = result.title,
                        category = result.category,
                        recordDate = result.recordDate,
                        localFilePath = result.localFilePath,
                        mimeType = result.mimeType
                    )
                    try {
                        repo.insertRecord(record)
                        _infoMessage.value = "Document imported successfully."
                    } catch (_: Exception) {
                        DocumentImportManager.deleteFile(result.localFilePath, context)
                        _errorMessage.value = "Failed to save record to local database."
                    }
                }
                is DocumentImportResult.Error -> {
                    _errorMessage.value = result.message
                }
            }
            _isImporting.value = false
        }
    }

    fun openRecord(record: MedicalRecord, context: Context) {
        val uid = authRepository?.currentUser?.uid
        val error = DocumentImportManager.openRecord(record, context, uid)
        if (error != null) {
            _errorMessage.value = error
        }
    }

    fun requestDeleteRecord(record: MedicalRecord) {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty() || record.accountUid != uid) {
            _errorMessage.value = "Unauthorized delete request."
            return
        }
        _recordToDelete.value = record
    }

    fun cancelDeleteRecord() {
        _recordToDelete.value = null
    }

    fun confirmDeleteRecord(context: Context) {
        val record = _recordToDelete.value ?: return
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty() || record.accountUid != uid) {
            _errorMessage.value = "Unauthorized delete request."
            _recordToDelete.value = null
            return
        }

        val repo = medicalRecordRepository ?: return
        viewModelScope.launch(dispatcherProvider.main) {
            try {
                repo.deleteRecord(record)
                withContext(Dispatchers.IO) {
                    DocumentImportManager.deleteFile(record.localFilePath, context, record, uid)
                }
                _infoMessage.value = "Record deleted."
            } catch (_: Exception) {
                _errorMessage.value = "Failed to delete record."
            } finally {
                _recordToDelete.value = null
            }
        }
    }
}

import androidx.compose.runtime.mutableIntStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.Modifier
import androidx.compose.ui.res.stringResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import com.example.abhayhealth.R
import com.example.abhayhealth.core.designsystem.components.EmptyStateContent

@Composable
fun TrendsScreen(
    viewModel: TrendsViewModel,
    modifier: Modifier = Modifier
) {
    var selectedTabIndex by remember { mutableIntStateOf(0) }
    val tabTitles = listOf(
        stringResource(id = R.string.trends_tab_labs),
        stringResource(id = R.string.trends_tab_vitals),
        stringResource(id = R.string.trends_tab_adherence)
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(horizontal = 16.dp)
    ) {
        Spacer(modifier = Modifier.height(16.dp))
        Text(
            text = stringResource(id = R.string.trends_title),
            style = MaterialTheme.typography.displayMedium,
            fontWeight = FontWeight.Bold,
            color = MaterialTheme.colorScheme.onBackground
        )
        Text(
            text = stringResource(id = R.string.trends_subtitle),
            style = MaterialTheme.typography.bodyLarge,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )
        Spacer(modifier = Modifier.height(16.dp))

        PrimaryTabRow(selectedTabIndex = selectedTabIndex) {
            tabTitles.forEachIndexed { index, title ->
                Tab(
                    selected = selectedTabIndex == index,
                    onClick = { selectedTabIndex = index },
                    text = { Text(text = title) }
                )
            }
        }

        Spacer(modifier = Modifier.height(16.dp))

        LazyColumn(
            contentPadding = PaddingValues(bottom = 16.dp),
            verticalArrangement = Arrangement.spacedBy(16.dp)
        ) {
            item {
                Card(
                    modifier = Modifier.fillMaxWidth(),
                    colors = CardDefaults.cardColors(
                        containerColor = MaterialTheme.colorScheme.surface
                    )
                ) {
                    when (selectedTabIndex) {
                        0 -> EmptyStateContent(
                            title = stringResource(id = R.string.trends_empty_labs_title),
                            description = stringResource(id = R.string.trends_empty_labs_desc),
                            icon = Icons.Default.Science
                        )
                        1 -> EmptyStateContent(
                            title = stringResource(id = R.string.trends_empty_vitals_title),
                            description = stringResource(id = R.string.trends_empty_vitals_desc),
                            icon = Icons.Default.MonitorHeart
                        )
                        else -> EmptyStateContent(
                            title = stringResource(id = R.string.trends_empty_adherence_title),
                            description = stringResource(id = R.string.trends_empty_adherence_desc),
                            icon = Icons.AutoMirrored.Filled.TrendingUp
                        )
                    }
                }
            }
        }
    }
}
