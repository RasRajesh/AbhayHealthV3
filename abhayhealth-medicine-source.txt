@Composable
fun MedicinesScreen(
    viewModel: MedicinesViewModel,
    modifier: Modifier = Modifier
) {
    val medications by viewModel.medications.collectAsState()
    var showAddDialog by remember { mutableStateOf(false) }

    if (showAddDialog) {
        AddMedicationDialog(
            onDismiss = { showAddDialog = false },
            onAdd = { name, dosage, slot ->
                viewModel.addMedication(name, dosage, slot)
                showAddDialog = false
            }
        )
    }

    LazyColumn(
        modifier = modifier
            .fillMaxSize()
            .padding(horizontal = 16.dp),
        contentPadding = PaddingValues(vertical = 16.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        item {
            Column {
                Text(
                    text = stringResource(id = R.string.meds_title),
                    style = MaterialTheme.typography.displayMedium,
                    fontWeight = FontWeight.Bold,
                    color = MaterialTheme.colorScheme.onBackground
                )
                Spacer(modifier = Modifier.height(4.dp))
                Text(
                    text = stringResource(id = R.string.meds_subtitle),
                    style = MaterialTheme.typography.bodyLarge,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }
        }

        item {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                Button(
                    onClick = { showAddDialog = true },
                    modifier = Modifier
                        .weight(1f)
                        .height(48.dp)
                ) {
                    Icon(imageVector = Icons.Default.Add, contentDescription = null)
                    Spacer(modifier = Modifier.width(4.dp))
                    Text(text = stringResource(id = R.string.meds_btn_add))
                }

                OutlinedButton(
                    onClick = { showAddDialog = true },
                    modifier = Modifier
                        .weight(1f)
                        .height(48.dp)
                ) {
                    Icon(imageVector = Icons.Default.Alarm, contentDescription = null)
                    Spacer(modifier = Modifier.width(4.dp))
                    Text(text = stringResource(id = R.string.meds_btn_reminders))
                }
            }
        }

        item {
            Text(
                text = stringResource(id = R.string.meds_section_schedule),
                style = MaterialTheme.typography.titleLarge,
                fontWeight = FontWeight.SemiBold,
                color = MaterialTheme.colorScheme.onBackground
            )
        }

        if (medications.isEmpty()) {
            item {
                Card(
                    modifier = Modifier.fillMaxWidth(),
                    colors = CardDefaults.cardColors(
                        containerColor = MaterialTheme.colorScheme.surface
                    )
                ) {
                    EmptyStateContent(
                        title = stringResource(id = R.string.meds_empty_schedule_title),
                        description = stringResource(id = R.string.meds_empty_schedule_desc),
                        icon = Icons.Default.Medication,
                        actionLabel = stringResource(id = R.string.meds_btn_add),
                        onActionClick = { showAddDialog = true }
                    )
                }
            }
        } else {
            val slots = listOf("MORNING (08:30)", "AFTERNOON (13:30)", "EVENING (19:00)", "NIGHT (21:30)")
            slots.forEach { slot ->
                val slotMeds = medications.filter { it.timeSlot.equals(slot, ignoreCase = true) }
                if (slotMeds.isNotEmpty()) {
                    item {
                        Text(
                            text = slot,
                            style = MaterialTheme.typography.titleMedium,
                            fontWeight = FontWeight.Bold,
                            color = MaterialTheme.colorScheme.primary
                        )
                    }
                    items(
                        items = slotMeds,
                        key = { it.id }
                    ) { med ->
                        MedicationCard(
                            medication = med,
                            onToggle = { viewModel.toggleDoseTaken(med.id) },
                            onDelete = { viewModel.removeMedication(med.id) }
                        )
                    }
                }
            }
        }
    }
}

@Composable
private fun MedicationCard(
    medication: PrescriptionMedication,
    onToggle: () -> Unit,
    onDelete: () -> Unit
) {
    Card(
        modifier = Modifier.fillMaxWidth(),
        colors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surface),
        elevation = CardDefaults.cardElevation(defaultElevation = 2.dp)
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            IconButton(onClick = onToggle) {
                Icon(
                    imageVector = if (medication.isTaken) Icons.Default.CheckCircle else Icons.Default.RadioButtonUnchecked,
                    contentDescription = if (medication.isTaken) "Dose taken" else "Mark dose taken",
                    tint = if (medication.isTaken) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurfaceVariant
                )
            }

            Spacer(modifier = Modifier.width(8.dp))

            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = medication.name,
                    style = MaterialTheme.typography.titleMedium,
                    fontWeight = FontWeight.SemiBold
                )
                Spacer(modifier = Modifier.height(2.dp))
                Text(
                    text = "${medication.dosage} â€¢ ${medication.timeSlot}",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }

            IconButton(onClick = onDelete) {
                Icon(
                    imageVector = Icons.Default.Delete,

        }
    }
}

@Composable
private fun AddMedicationDialog(
    onDismiss: () -> Unit,
    onAdd: (String, String, String) -> Unit
) {
    var name by remember { mutableStateOf("") }
    var dosage by remember { mutableStateOf("") }
    var selectedSlot by remember { mutableStateOf("MORNING (08:30)") }

    val slots = listOf("MORNING (08:30)", "AFTERNOON (13:30)", "EVENING (19:00)", "NIGHT (21:30)")

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("Add Prescription Medicine") },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                OutlinedTextField(
                    value = name,
                    onValueChange = { name = it },
                    label = { Text("Medicine Name (e.g. Paracetamol)") },
                    singleLine = true
                )
                OutlinedTextField(
                    value = dosage,
                    onValueChange = { dosage = it },
                    label = { Text("Dosage (e.g. 500 mg - 1 Tablet)") },
                    singleLine = true
                )
                Text("Schedule Slot", style = MaterialTheme.typography.labelLarge)
                slots.forEach { slot ->
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(40.dp)
                    ) {
                        Checkbox(
                            checked = selectedSlot == slot,
                            onCheckedChange = { if (it) selectedSlot = slot }
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(slot, style = MaterialTheme.typography.bodyMedium)
                    }
                }
            }
        },
        confirmButton = {
            Button(
                onClick = { onAdd(name, dosage, selectedSlot) },
                enabled = name.isNotBlank()
            ) {
                Text("Add Medicine")
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text("Cancel")
            }
        }
    )
}

package com.example.abhayhealth.feature.medicines

import androidx.lifecycle.ViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.UUID

data class PrescriptionMedication(
    val id: String = UUID.randomUUID().toString(),
    val name: String,
    val dosage: String,
    val timeSlot: String, // MORNING (08:30), AFTERNOON (13:30), EVENING (19:00), NIGHT (21:30)
    val isTaken: Boolean = false
)

class MedicinesViewModel : ViewModel() {

    private val _medications = MutableStateFlow<List<PrescriptionMedication>>(emptyList())
    val medications: StateFlow<List<PrescriptionMedication>> = _medications.asStateFlow()

    fun addMedication(name: String, dosage: String, timeSlot: String) {
        if (name.isBlank()) return
        val current = _medications.value.toMutableList()
        current.add(
            PrescriptionMedication(
                name = name.trim(),
                dosage = dosage.trim().ifBlank { "1 Tablet" },
                timeSlot = timeSlot
            )
        )
        _medications.value = current
    }

    fun toggleDoseTaken(id: String) {
        _medications.value = _medications.value.map { med ->
            if (med.id == id) med.copy(isTaken = !med.isTaken) else med
        }
    }

    fun removeMedication(id: String) {
        _medications.value = _medications.value.filter { it.id != id }
    }
}

package com.example.abhayhealth.feature.records

import android.content.Context
import android.net.Uri
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.abhayhealth.core.common.DefaultDispatcherProvider
import com.example.abhayhealth.core.common.DispatcherProvider
import com.example.abhayhealth.domain.model.AuthState
import com.example.abhayhealth.domain.repository.AuthRepository
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.collectLatest
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class RecordsViewModel(
    private val medicalRecordRepository: MedicalRecordRepository? = null,
    private val authRepository: AuthRepository? = null,
    private val dispatcherProvider: DispatcherProvider = DefaultDispatcherProvider()
) : ViewModel() {

    private val _records = MutableStateFlow<List<MedicalRecord>>(emptyList())
    val records: StateFlow<List<MedicalRecord>> = _records.asStateFlow()

    private val _searchQuery = MutableStateFlow("")
    val searchQuery: StateFlow<String> = _searchQuery.asStateFlow()

    private val _isImporting = MutableStateFlow(false)
    val isImporting: StateFlow<Boolean> = _isImporting.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private val _infoMessage = MutableStateFlow<String?>(null)
    val infoMessage: StateFlow<String?> = _infoMessage.asStateFlow()

    private val _recordToDelete = MutableStateFlow<MedicalRecord?>(null)
    val recordToDelete: StateFlow<MedicalRecord?> = _recordToDelete.asStateFlow()

    private var observeJob: Job? = null

    init {
        observeAuthState()
    }

    private fun observeAuthState() {
        val authRepo = authRepository ?: return
        viewModelScope.launch(dispatcherProvider.main) {
            authRepo.authState.collectLatest { state ->
                when (state) {
                    is AuthState.Authenticated -> {
                        observeRecordsForUid(state.user.uid)
                    }
                    else -> {
                        clearAllState()
                    }
                }
            }
        }
    }

    fun observeRecords() {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty()) {
            clearAllState()
        } else {
            observeRecordsForUid(uid)
        }
    }

    private fun observeRecordsForUid(uid: String) {
        val repo = medicalRecordRepository ?: return
        observeJob?.cancel()
        observeJob = viewModelScope.launch(dispatcherProvider.main) {
            if (_searchQuery.value.isBlank()) {
                repo.getRecords(uid).collectLatest { list ->
                    _records.value = list
                }
            } else {
                repo.searchRecords(uid, _searchQuery.value).collectLatest { list ->
                    _records.value = list
                }
            }
        }
    }

    private fun clearAllState() {
        observeJob?.cancel()
        _records.value = emptyList()
        _searchQuery.value = ""
        _errorMessage.value = null
        _infoMessage.value = null
        _recordToDelete.value = null
    }

    fun onSearchQueryChanged(query: String) {
        _searchQuery.value = query
        val uid = authRepository?.currentUser?.uid ?: return
        observeRecordsForUid(uid)
    }

    fun clearMessages() {
        _errorMessage.value = null
        _infoMessage.value = null
    }

    fun importDocument(uri: Uri, context: Context) {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty()) {
            _errorMessage.value = "Please sign in to import medical documents."
            return
        }

        val repo = medicalRecordRepository
        if (repo == null) {
            _errorMessage.value = "Local database unavailable."
            return
        }

        viewModelScope.launch(dispatcherProvider.main) {
            _isImporting.value = true
            _errorMessage.value = null

            val result = withContext(Dispatchers.IO) {
                DocumentImportManager.importDocument(uri, context)
            }

            when (result) {
                is DocumentImportResult.Success -> {
                    val record = MedicalRecord(
                        accountUid = uid,
                        title = result.title,
                        category = result.category,
                        recordDate = result.recordDate,
                        localFilePath = result.localFilePath,
                        mimeType = result.mimeType
                    )
                    try {
                        repo.insertRecord(record)
                        _infoMessage.value = "Document imported successfully."
                    } catch (_: Exception) {
                        DocumentImportManager.deleteFile(result.localFilePath, context)
                        _errorMessage.value = "Failed to save record to local database."
                    }
                }
                is DocumentImportResult.Error -> {
                    _errorMessage.value = result.message
                }
            }
            _isImporting.value = false
        }
    }

    fun openRecord(record: MedicalRecord, context: Context) {
        val uid = authRepository?.currentUser?.uid
        val error = DocumentImportManager.openRecord(record, context, uid)
        if (error != null) {
            _errorMessage.value = error
        }
    }

    fun requestDeleteRecord(record: MedicalRecord) {
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty() || record.accountUid != uid) {
            _errorMessage.value = "Unauthorized delete request."
            return
        }
        _recordToDelete.value = record
    }

    fun cancelDeleteRecord() {
        _recordToDelete.value = null
    }

    fun confirmDeleteRecord(context: Context) {
        val record = _recordToDelete.value ?: return
        val uid = authRepository?.currentUser?.uid
        if (uid.isNullOrEmpty() || record.accountUid != uid) {
            _errorMessage.value = "Unauthorized delete request."
            _recordToDelete.value = null
            return
        }

        val repo = medicalRecordRepository ?: return
        viewModelScope.launch(dispatcherProvider.main) {
            try {
                repo.deleteRecord(record)
                withContext(Dispatchers.IO) {
                    DocumentImportManager.deleteFile(record.localFilePath, context, record, uid)
                }
                _infoMessage.value = "Record deleted."
            } catch (_: Exception) {
                _errorMessage.value = "Failed to delete record."
            } finally {
                _recordToDelete.value = null
            }
        }
    }
}
